cmake_minimum_required(VERSION 3.22)
project(PejededStudio VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_OSX_DEPLOYMENT_TARGET "12.0" CACHE STRING "Oldest macOS")

add_subdirectory(third_party/JUCE)
add_subdirectory(third_party/Catch2)

juce_add_gui_app(PejededStudio PRODUCT_NAME "PejededStudio")
set_target_properties(PejededStudio PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR})
target_sources(PejededStudio PRIVATE src/app/Main.cpp)
target_include_directories(PejededStudio PRIVATE src)
target_compile_definitions(PejededStudio PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_APPLICATION_NAME_STRING="PEJEDED STUDIO" JUCE_APPLICATION_VERSION_STRING="0.1.0")
target_link_libraries(PejededStudio PRIVATE pejeded_core pejeded_app juce::juce_gui_basics juce::juce_audio_devices juce::juce_audio_utils juce::juce_opengl
  PUBLIC juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)

enable_testing()
add_library(pejeded_core STATIC src/app/AppInfo.cpp src/emulators/Pattern.cpp src/emulators/Mc707Engine.cpp src/emulators/PatternReloader.cpp src/app/Console.cpp src/store/Store.cpp)
add_library(sqlite3 STATIC third_party/sqlite/sqlite3.c)
target_include_directories(sqlite3 PUBLIC third_party/sqlite)
target_compile_definitions(sqlite3 PUBLIC SQLITE_ENABLE_DBSTAT_VTAB=1 SQLITE_DQS=0 SQLITE_THREADSAFE=1)
set_target_properties(sqlite3 PROPERTIES POSITION_INDEPENDENT_CODE ON)
find_package(Threads REQUIRED)
target_link_libraries(sqlite3 PUBLIC Threads::Threads ${CMAKE_DL_LIBS})
target_include_directories(pejeded_core PUBLIC src)
target_link_libraries(pejeded_core PUBLIC sqlite3)
if(NOT MSVC)
  target_compile_options(pejeded_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(unit_tests tests/test_app.cpp tests/test_pattern.cpp tests/test_engine.cpp tests/test_reload.cpp tests/test_console.cpp tests/test_store.cpp tests/test_settings_file.cpp tests/test_start_check.cpp tests/test_master_limiter.cpp tests/test_device_life.cpp)
target_compile_definitions(unit_tests PRIVATE PEJEDED_TEST_DATA="${CMAKE_CURRENT_SOURCE_DIR}/tests/data" PEJEDED_RESOURCES="${CMAKE_CURRENT_SOURCE_DIR}/resources")
target_include_directories(unit_tests PRIVATE src)
target_link_libraries(unit_tests PRIVATE pejeded_core Catch2::Catch2WithMain)
if(NOT MSVC)
  target_compile_options(unit_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
# JUCE-based app pieces that are tested without a window
add_library(pejeded_app STATIC src/app/SettingsFile.cpp src/app/StartCheck.cpp)
target_link_libraries(pejeded_app PUBLIC pejeded_core juce::juce_core)
target_compile_definitions(pejeded_app PUBLIC JUCE_STANDALONE_APPLICATION=1 JUCE_USE_CURL=0 JUCE_WEB_BROWSER=0 JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1)
target_link_libraries(unit_tests PRIVATE pejeded_app)
include(${CMAKE_CURRENT_SOURCE_DIR}/third_party/Catch2/extras/Catch.cmake)
catch_discover_tests(unit_tests)
add_test(NAME e2e_setup_wizard COMMAND bash ${CMAKE_CURRENT_SOURCE_DIR}/tests/e2e_wizard.sh $<TARGET_FILE:PejededStudio>)
add_test(NAME e2e_exec_quit COMMAND bash ${CMAKE_CURRENT_SOURCE_DIR}/tests/e2e_exec_quit.sh $<TARGET_FILE:PejededStudio>)
